imie VARCHAR (40) NOT NULL,
nazwisko VARCHAR (60) NOT NULL,
data_urodzenia DATE NOT NULL);

CREATE TABLE zadanie(
id_zadania SERIAL auto_increment PRIMARY KEY,
nazwa_zadania VARCHAR (150) NOT NULL,
priorytet VARCHAR (10) CHECK (priorytet IN ("normalny","wysoki","niski")),
opis TEXT,
pracownik INT,
CONSTRAINT fk_pracownik FOREIGN KEY (pracownik)
REFERENCES pracownik(id_pracownika)
ON DELETE SET NULL);
